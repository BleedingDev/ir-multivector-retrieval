    93	class BertSelfAttention(nn.Module):
    94	    """Multi-head self-attention block matching HF BertSelfAttention's
    95	    weight layout exactly. q/k/v are separate Linear layers (HF stores
    96	    them split, not as a fused QKV)."""
    97	
    98	    def __init__(self, hidden: int, n_heads: int):
    99	        super().__init__()
   100	        if hidden % n_heads != 0:
   101	            raise SystemExit(f"hidden {hidden} not divisible by n_heads {n_heads}")
   102	        self.n_heads = n_heads
   103	        self.head_dim = hidden // n_heads
   104	        self.q = nn.Linear(hidden, hidden)
   105	        self.k = nn.Linear(hidden, hidden)
   106	        self.v = nn.Linear(hidden, hidden)
   107	        self.out = nn.Linear(hidden, hidden)
   108	        self.ln = nn.LayerNorm(hidden)
   109	
   110	    def __call__(self, x: mx.array, attn_bias: mx.array) -> mx.array:
   111	        B, T, H = x.shape
   112	        n = self.n_heads
   113	        d = self.head_dim
   114	        scale = 1.0 / (d ** 0.5)
   115	
   116	        q = self.q(x).reshape(B, T, n, d).transpose(0, 2, 1, 3)  # [B,n,T,d]
   117	        k = self.k(x).reshape(B, T, n, d).transpose(0, 2, 1, 3)
   118	        v = self.v(x).reshape(B, T, n, d).transpose(0, 2, 1, 3)
   119	
   120	        # scores [B, n, T, T]
   121	        scores = (q @ k.transpose(0, 1, 3, 2)) * scale
   122	        scores = scores + attn_bias  # broadcast [B,1,1,T]
   123	        attn = mx.softmax(scores, axis=-1)
   124	        ctx = attn @ v  # [B, n, T, d]
   125	        ctx = ctx.transpose(0, 2, 1, 3).reshape(B, T, H)
   126	
   127	        # output projection + residual + LN (BERT post-LN)
   128	        return self.ln(self.out(ctx) + x)

   131	class BertFFN(nn.Module):
   132	    """Position-wise FFN: up-project (gelu) → down-project → residual + LN."""
   133	
   134	    def __init__(self, hidden: int, intermediate: int):
   135	        super().__init__()
   136	        self.up = nn.Linear(hidden, intermediate)
   137	        self.down = nn.Linear(intermediate, hidden)
   138	        self.ln = nn.LayerNorm(hidden)
   139	
   140	    def __call__(self, x: mx.array) -> mx.array:
   141	        h = self.up(x)
   142	        h = nn.gelu(h)
   143	        h = self.down(h)
   144	        return self.ln(h + x)

   154	        x = self.attn(x, attn_bias)

   159	class BertEmbeddings(nn.Module):
   160	    def __init__(self, vocab: int, max_pos: int, type_vocab: int, hidden: int):
   161	        super().__init__()
   162	        self.word = nn.Embedding(vocab, hidden)
   163	        self.position = nn.Embedding(max_pos, hidden)
   164	        self.token_type = nn.Embedding(type_vocab, hidden)
   165	        self.ln = nn.LayerNorm(hidden)
   166	
   167	    def __call__(self, input_ids: mx.array, token_type_ids: mx.array) -> mx.array:
   168	        T = input_ids.shape[1]
   169	        pos_ids = mx.arange(T, dtype=mx.int32)
   170	        # broadcast pos_ids over batch
   171	        x = self.word(input_ids) + self.position(pos_ids) + self.token_type(token_type_ids)

   175	class ColBertMLX(nn.Module):
   176	    """BERT base + final linear projection (no bias) + optional L2-normalize.
   177	    Forward returns the per-token L2-normalised projected embeddings."""
   178	
   179	    def __init__(self, cfg: dict):
   180	        super().__init__()
   181	        self.cfg = cfg
   182	        self.embeddings = BertEmbeddings(
   183	            vocab=cfg["vocab_size"],
   184	            max_pos=cfg["max_position_embeddings"],
   185	            type_vocab=cfg["type_vocab_size"],
   186	            hidden=cfg["hidden_size"],
   187	        )
   188	        self.layers = [
   189	            BertLayer(
   190	                hidden=cfg["hidden_size"],
   191	                n_heads=cfg["n_heads"],
   192	                intermediate=cfg["intermediate_size"],
   193	            )
   194	            for _ in range(cfg["n_layers"])
   195	        ]
   196	        self.projection = nn.Linear(
   197	            cfg["projection_in"], cfg["projection_out"], bias=cfg["projection_has_bias"]
   198	        )
   199	
   200	    def __call__(
   201	        self,
   202	        input_ids: mx.array,
   203	        attention_mask: mx.array,
   204	        token_type_ids: mx.array,
   205	    ) -> mx.array:
   206	        # attention bias: 0 where mask=1, -inf where mask=0 (so softmax→0).
   207	        # shape: [B, 1, 1, T] to broadcast over (n_heads, query_seq).
   208	        bias = (1.0 - attention_mask.astype(input_ids.dtype)).astype(mx.float32) * -1e4
   209	        bias = bias.reshape(bias.shape[0], 1, 1, bias.shape[1])
   210	
   211	        x = self.embeddings(input_ids, token_type_ids)
   212	        # Cast bias to x's dtype so attention math stays in compute precision.
   213	        bias = bias.astype(x.dtype)
   214	
   215	        for layer in self.layers:
   216	            x = layer(x, bias)
   217	
   218	        # 768 -> 128 projection
   219	        x = self.projection(x)
   220	        # L2-normalize per token along last axis
   221	        norm = mx.sqrt(mx.sum(x * x, axis=-1, keepdims=True) + 1e-12)
   222	        return x / norm

   296	def read_docs_jsonl(path: Path, max_docs: int | None) -> list[dict]:
   297	    out: list[dict] = []
   298	    with path.open("r", encoding="utf-8") as f:
   299	        for lineno, line in enumerate(f, start=1):
   300	            line = line.strip()
   301	            if not line:
   302	                continue
   303	            try:
   304	                obj = json.loads(line)
   305	            except json.JSONDecodeError as e:
   306	                raise SystemExit(f"{path}:{lineno}: malformed JSON: {e}") from e
   307	            if "doc_id" not in obj or "text" not in obj:
   308	                raise SystemExit(
   309	                    f"{path}:{lineno}: each line needs 'doc_id' and 'text' fields"
   310	                )
   311	            out.append(obj)
   312	            if max_docs is not None and len(out) >= max_docs:
   313	                break
   314	    return out

   317	def encode_docs_mlx(
   318	    docs: list[dict],
   319	    *,
   320	    model_name: str,
   321	    weights_path: Path,
   322	    dtype: str,
   323	    batch_size: int,
   324	    trust_remote_code: bool,
   325	) -> tuple[list[EncodedDoc], list[tuple[str, str]], int]:
   326	    """Encode docs through MLX. Returns (encoded, dropped, dim).
   327	
   328	    Tokenizer + skiplist come from pylate (CPU, deterministic, fast).
   329	    Forward pass + L2-norm run in MLX. Output rows match the keep-mask
   330	    (skiplist & attention_mask), exactly as encode.py does.
   331	    """
   332	    # Tokenizer comes from pylate. We don't run model.forward through pylate
   333	    # — only model.tokenize and model.skiplist_mask.
   334	    from pylate import models  # type: ignore
   335	
   336	    pl = models.ColBERT(
   337	        model_name_or_path=model_name,
   338	        trust_remote_code=trust_remote_code,
   339	        device="cpu",
   340	    )
   341	    pl.eval()
   342	    skiplist = pl.skiplist  # list[int]
   343	
   344	    # Read MLX safetensors + config.
   345	    if not weights_path.exists():
   346	        raise SystemExit(
   347	            f"weights file {weights_path} missing — run tools/export_colbert_to_mlx.py first"
   348	        )

   350	    if not config_path.exists():
   351	        raise SystemExit(f"config.json missing next to {weights_path}")

   356	    # Cast weights to compute dtype so the forward pass is consistent.
   357	    weights = {k: v.astype(compute_dtype) for k, v in weights.items()}
   358	
   359	    model = ColBertMLX(cfg)
   360	    _load_weights_into_model(model, weights)

   363	    skip_set = set(int(s) for s in skiplist)

   366	    dropped: list[tuple[str, str]] = []
   367	    dim = cfg["projection_out"]

   369	    for batch_start in range(0, len(docs), batch_size):
   370	        batch = docs[batch_start : batch_start + batch_size]
   371	        texts = [d["text"] for d in batch]
   372	        try:
   373	            features = pl.tokenize(texts, is_query=False)
   374	            input_ids_pt = features["input_ids"]
   375	            attn_pt = features["attention_mask"]

   381	                type_ids_pt = torch.zeros_like(input_ids_pt)

   387	            tok_emb = model(input_ids, attn, type_ids)  # [B, T, dim]
   388	            # Force evaluation so we have concrete arrays for slicing.
   389	            mx.eval(tok_emb)
   390	        except Exception as e:  # noqa: BLE001
   391	            for d in batch:
   392	                dropped.append((str(d["doc_id"]), f"batch encode failed: {e!r}"))
   393	            continue
   394	
   395	        # Apply keep mask (skiplist & attention_mask) per row.
   396	        ids_np = input_ids_pt.cpu().numpy()

   399	        # mlx.array → host: prefer mx.eval + .tolist or numpy() if available
   400	        if hasattr(emb_np, "__array__"):

   402	
   403	            emb_arr = mx.eval(emb_np)
   404	            # mlx.core arrays support np.array(arr) conversion
   405	            import numpy as np
   406	
   407	            emb_arr = np.array(emb_np, copy=False)
   408	        else:
   409	            import numpy as np
   410	
   411	            emb_arr = np.array(emb_np.tolist(), dtype=np.float32)
   412	
   413	        for b, d in enumerate(batch):

   416	                row_attn = attn_np[b]
   417	                kept_idx = [
   418	                    i
   419	                    for i in range(len(row_ids))
   420	                    if row_attn[i] == 1 and int(row_ids[i]) not in skip_set
   421	                ]
   422	                if not kept_idx:
   423	                    dropped.append((str(d["doc_id"]), "0-token output after masking"))
   424	                    continue
   425	                kept_ids = [int(row_ids[i]) for i in kept_idx]

   436	            except Exception as e:  # noqa: BLE001

   250	def gate2_cosine(pt: dict, mlx: dict, mean_floor: float, min_floor: float) -> tuple[bool, str]:
   251	    cos = cosines_per_token(pt, mlx)
   252	    cos_sorted = sorted(cos)
   253	    n = len(cos)
   254	    mean = sum(cos) / n

   268	def gate3_determinism(out_a: Path, out_b: Path, max_docs: int | None, dtype: str) -> tuple[bool, str]:
   269	    encode_mlx(out_a, max_docs, dtype)
   270	    encode_mlx(out_b, max_docs, dtype)
   271	    a = parse_tokens_bin(out_a)
   272	    b = parse_tokens_bin(out_b)
   273	    if a["ids"] != b["ids"] or a["offsets"] != b["offsets"]:
   274	        return False, "two-run determinism: token_ids/offsets differ"
   275	
   276	    # Compare floats with strict byte-equality first; if that fails, fall
   277	    # back to ULP-class threshold (≤ 4 ULPs ≈ ~5e-7 abs for fp32 values
   278	    # near 0.1, OK for L2-normalised ColBERT vectors).
   279	    diffs = 0
   280	    max_abs = 0.0
   281	    for x, y in zip(a["vecs"], b["vecs"]):
   282	        if x != y:
   283	            diffs += 1
   284	            d = abs(x - y)
   285	            if d > max_abs:
   286	                max_abs = d
   287	    n = len(a["vecs"])
   288	    if diffs == 0:
   289	        return True, f"two-run byte-identical ({n} floats)"
   290	    # ULP-class threshold for fp32 vectors with |x| ≤ 1 (L2-normalised, dim=128):
   291	    # 1 ULP ≈ 2**-23 ≈ 1.19e-7. Allow ≤ 4 ULPs.
   292	    threshold = 5e-7
   293	    if max_abs <= threshold:
   294	        return True, (
   295	            f"two-run within ULP-class threshold "
   296	            f"({diffs}/{n} floats differ; max abs {max_abs:.2e} ≤ {threshold:.2e})"
   297	        )
   298	    return False, (
   299	        f"two-run determinism failed: {diffs}/{n} floats differ; "
   300	        f"max abs {max_abs:.2e} > {threshold:.2e}"
   301	    )

   304	def gate4_clustering(pt_bin: Path, mlx_bin: Path, kappa: int, seed: int) -> tuple[bool, str]:
   305	    pt_tac = pt_bin.with_suffix(".pt.tac")
   306	    mlx_tac = mlx_bin.with_suffix(".mlx.tac")

   312	
   313	    # Header agreement on kappa/dim/n_*.
   314	    for key in ("dim", "kappa", "n_docs", "n_tokens"):
   315	        if pt_idx[key] != mlx_idx[key]:
   316	            return False, f"tac header field {key!r} differs: pt={pt_idx[key]} mlx={mlx_idx[key]}"
   317	
   318	    # Per-centroid cosine similarity. Centroid c maps the same vocab id

   326	    cos_per_centroid = []
   327	    for i in range(kappa_v):
   328	        a = pt_c[i * dim : (i + 1) * dim]
   329	        b = mlx_c[i * dim : (i + 1) * dim]
   330	        cos_per_centroid.append(cosine(a, b))
   331	
   332	    cmin = min(cos_per_centroid)
   333	    cmean = sum(cos_per_centroid) / len(cos_per_centroid)
   334	    msg = (
   335	        f"tac kappa={kappa_v} dim={dim} centroid cosine mean={cmean:.6f} min={cmin:.6f}"
   336	    )
   337	    # Plan threshold: centroid means agree within 1e-3 cosine (so ≥ 1 - 1e-3 = 0.999).
   338	    if cmin < 0.999:
   339	        return False, msg + f" — min < 0.999"
   340	    return True, msg

   ```python
   ctx = mx.fast.scaled_dot_product_attention(q, k, v, scale=scale, mask=attn_bias)
   ```
   then keep the existing transpose/reshape/output projection.

   Effort: S.  
   Rationale: this is the most obvious Apple Silicon hot-path miss and should reduce attention kernel count and memory traffic.

   Evidence: [tools/export_colbert_to_mlx.py](/Users/satan/side/experiments/ir-multivector-retrieval/tools/export_colbert_to_mlx.py:85) already supports fp16 export, but [tools/encode_mlx.py](/Users/satan/side/experiments/ir-multivector-retrieval/tools/encode_mlx.py:357) always `astype`s every tensor to runtime dtype.

   Impact: optimistic 5-15% on 100-doc CLI wall; realistic 3-8%. Jira <3%.  
   Effort: S/M.  
   Rationale: avoid allocating and converting the whole BERT weight set during small jobs.

   Effort: M.  
   Rationale: BERT attention cost grows with padded sequence length, so packing similar lengths is high ROI.
